function calculateGP() { hideError(); // --------------------------------------------- // 1. Check Product // --------------------------------------------- if (product.value === "") { showError("Please choose a product."); return; } // --------------------------------------------- // 2. Get TP // --------------------------------------------- const tpValue = cleanNumber(tp.value); if ( tpValue === null || tpValue < 0 ) { showError("Please check TP."); return; } // --------------------------------------------- // 3. Read Bonus Qty // Example: 10+2 // Purchased = 10 // Free = 2 // --------------------------------------------- let bonusText = bonusQty.value .trim() .replace(/\s/g, ""); if (bonusText === "") { showError( "Please enter Bonus Qty, for example 10+2." ); return; } // Allow formats such as: // 10+2 // 12+2 // 20+5 const bonusParts = bonusText.split("+"); if (bonusParts.length !== 2) { showError( "Please enter Bonus Qty in the format 10+2." ); return; } const purchasedQty = parseFloat(bonusParts[0]); const freeQty = parseFloat(bonusParts[1]); if ( isNaN(purchasedQty) || isNaN(freeQty) ) { showError( "Please enter valid Bonus Qty, for example 10+2." ); return; } if (purchasedQty <= 0) { showError( "Purchased Qty must be greater than zero." ); return; } if (freeQty < 0) { showError( "Free Qty cannot be negative." ); return; } // --------------------------------------------- // 4. Calculate Free Qty % // --------------------------------------------- const freeQtyPercent = (freeQty / purchasedQty) * 100; bonusPercent.value = freeQtyPercent.toFixed(2); // --------------------------------------------- // 5. Get MRP and GST // --------------------------------------------- const mrpValue = cleanNumber(mrp.value); const gstValue = cleanNumber(gst.value); if ( mrpValue === null || mrpValue <= 0 ) { showError( "Please select a product with valid MRP." ); return; } if ( gstValue === null || gstValue < 0 ) { showError( "Please select a product with valid GST." ); return; } // --------------------------------------------- // 6. Calculate PTS // // PTS = // MRP / (1 + GST%) // × 80% // × 90% // --------------------------------------------- const ptsValue = ( mrpValue / (1 + gstValue / 100) ) * 0.80 * 0.90; pts.value = ptsValue.toFixed(2); // --------------------------------------------- // 7. Calculate Effective TP // // Example: // // TP = ₹100 // Bonus = 10+2 // Free Qty % = 20% // // Effective TP = // 100 × 1.20 // = ₹120 // --------------------------------------------- const effectiveTP = tpValue * ( 1 + freeQtyPercent / 100 ); // --------------------------------------------- // 8. Calculate GP ₹ // // GP = PTS - Effective TP // --------------------------------------------- const gp = ptsValue - effectiveTP; // --------------------------------------------- // 9. Calculate GP % // // GP % = GP / PTS × 100 // --------------------------------------------- const gpPercent = ( gp / ptsValue ) * 100; // --------------------------------------------- // 10. SHOW RESULTS // --------------------------------------------- resultPts.textContent = money(ptsValue); resultBonusPercent.textContent = freeQtyPercent.toFixed(2) + "%"; resultEffectiveTP.textContent = money(effectiveTP); resultGP.textContent = money(gp); resultGPPercent.textContent = gpPercent.toFixed(2) + "%"; // --------------------------------------------- // 11. Optional: scroll to result on mobile // --------------------------------------------- document .querySelector(".result") .scrollIntoView({ behavior: "smooth", block: "nearest" }); }
